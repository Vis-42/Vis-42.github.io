# ╔═╡ c3d4e5f6-0003-11f1-0004-000000000003
begin
    # ── Clique complex generation ──────────────────────────────────────────
    # Build a clique complex from an Erdos-Renyi random graph G(N, p_edge).
    # A clique complex (also called the Vietoris-Rips complex at the clique level)
    # promotes every complete subgraph (clique) in the underlying graph to a simplex.
    # Triangles and tetrahedra are therefore determined entirely by the edge structure
    # rather than being independently prescribed: no extra randomness is needed beyond
    # the edge coin flips, which makes the construction fully reproducible from one seed.
    function _gen_complex(N::Int, p_edge::Float64, seed::Int)
        # Fix the RNG seed so the same (N, p_edge) always produces the same graph.
        # This is important: the complex is built once and held fixed while the
        # percolation probability p is swept, so changes in the S(p) curve come
        # from different occupation realisations, not from a different graph.
        rng = Random.MersenneTwister(seed)

        # adj[i,j] = true means the edge {i,j} exists in the base graph.
        # Storing the full N×N Boolean matrix allows O(1) adjacency lookups
        # during the triangle and tetrahedron enumeration below.
        adj = falses(N, N)

        # edges collects every edge as an ordered pair (i < j) in the upper triangle.
        # Keeping this list avoids scanning all N^2 pairs again when building triangles.
        edges = Tuple{Int,Int}[]

        # G(N, p_edge) Erdos-Renyi model: each of the C(N,2) candidate edges is
        # included independently with probability p_edge.
        # Restricting to i < j means each pair is tested exactly once.
        for i in 1:N, j in i+1:N
            if rand(rng) < p_edge
                # Write both directions so adjacency lookups are symmetric.
                adj[i,j]=adj[j,i]=true; push!(edges,(i,j))
            end
        end

        # ── 2-simplices (triangles) ──────────────────────────────────────
        # For each existing edge (i,j), test every vertex k > j as a potential
        # third corner. The triple (i,j,k) forms a triangle iff {i,k} and {j,k}
        # are both present. This is the clique-extension step: we try to extend
        # a 1-simplex (edge) to a 2-simplex (triangle) by one vertex.
        # Restricting k > j ensures each triangle is listed exactly once in
        # sorted order (i < j < k).
        triangles = Tuple{Int,Int,Int}[]
        for (i,j) in edges, k in j+1:N
            adj[i,k]&&adj[j,k]&&push!(triangles,(i,j,k))
        end

        # ── 3-simplices (tetrahedra) ─────────────────────────────────────
        # Extend each triangle (i,j,k) by a fourth vertex l > k that is adjacent
        # to ALL three existing vertices. This requires three adjacency checks
        # (adj[i,l], adj[j,l], adj[k,l]), confirming that l closes a K_4 clique.
        # Again, l > k guarantees each tetrahedron appears once in sorted order.
        tets = Tuple{Int,Int,Int,Int}[]
        for (i,j,k) in triangles, l in k+1:N
            adj[i,l]&&adj[j,l]&&adj[k,l]&&push!(tets,(i,j,k,l))
        end

        # Pack everything into a named tuple so downstream functions can access
        # cx.N, cx.adj, cx.edges, cx.triangles, cx.tets by name.
        (N=N, adj=adj, edges=edges, triangles=triangles, tets=tets)
    end

    # ── Single percolation realisation ────────────────────────────────────
    # Perform one independent bond-percolation trial on the simplicial complex cx
    # at occupation probability p, acting on simplices of the chosen dimension dim.
    # Each call uses its own RNG seed so that n_rep calls at the same p produce
    # statistically independent samples (needed for computing variance in _sweep).
    # Returns S = (size of largest connected component) / N, the order parameter.
    function _perc_once(cx, p::Float64, dim::Int, seed::Int)
        N=cx.N
        # Per-realisation seed: seed=1,2,...,n_rep gives independent samples.
        rng=Random.MersenneTwister(seed)

        # adj_p is the adjacency matrix of the percolated subgraph.
        # It starts all-false (no bonds open) and is filled in according to dim.
        # Keeping it as a dense Boolean matrix mirrors the structure of cx.adj,
        # making the BFS below identical across all three cases.
        adj_p=falses(N,N)

        if dim==1
            # Standard (pairwise) bond percolation: flip an independent coin for
            # each edge; open edges join their two endpoints in adj_p.
            # This is the classical Erdos-Renyi percolation process on the graph skeleton.
            for (i,j) in cx.edges; rand(rng)<p&&(adj_p[i,j]=adj_p[j,i]=true); end

        elseif dim==2
            # Triangle (2-simplex) percolation: the unit of occupation is a full triangle.
            # When a triangle is occupied, ALL THREE of its edges activate simultaneously.
            # This is what makes the transition cooperative: two vertices i and j are
            # connected through a triangle only if the entire triangle is present,
            # not merely the edge {i,j}. As a result, the effective percolation threshold
            # is higher than for dim=1, and for dense complexes the rise in S(p) can be
            # abrupt (first-order-like) rather than gradual.
            for (i,j,k) in cx.triangles
                rand(rng)<p&&(adj_p[i,j]=adj_p[j,i]=adj_p[i,k]=adj_p[k,i]=adj_p[j,k]=adj_p[k,j]=true)
            end

        elseif dim==3
            # Tetrahedron (3-simplex) percolation: the unit is a full tetrahedron.
            # Occupation simultaneously opens all C(4,2)=6 edges of the tetrahedron.
            # The cooperative constraint is even stronger than for triangles: a
            # conducting path must thread through entire K_4 subgraphs, raising p_c further.
            for (i,j,k,l) in cx.tets
                if rand(rng)<p
                    # Enumerate all six vertex-pairs of the tetrahedron explicitly.
                    # Any occupied tet writes these six edges into adj_p.
                    for (a,b) in [(i,j),(i,k),(i,l),(j,k),(j,l),(k,l)]
                        adj_p[a,b]=adj_p[b,a]=true
                    end
                end
            end
        end

        # ── BFS connected-component labeling ────────────────────────────
        # Walk the percolated graph adj_p and assign a unique integer ID to each
        # connected component. comp[v]=0 means vertex v has not been visited yet.
        comp=zeros(Int,N); c=0
        for start in 1:N
            # Skip vertices already assigned to a component.
            comp[start]!=0&&continue
            # Start a new component and seed the BFS queue with this vertex.
            c+=1; queue=[start]; comp[start]=c
            while !isempty(queue)
                # popfirst! gives FIFO order (true BFS); for component labeling,
                # DFS would give the same result but BFS avoids deep call stacks.
                v=popfirst!(queue)
                # For each unvisited neighbor of v, assign the same component ID
                # and enqueue it. adj_p[v,u] is O(1) via the dense matrix.
                # The O(N) inner loop is acceptable for N <= 200.
                for u in 1:N; adj_p[v,u]&&comp[u]==0&&(comp[u]=c;push!(queue,u)); end
            end
        end

        # ── Giant-component size ─────────────────────────────────────────
        # Count the number of vertices in each of the c components, then
        # return the largest count normalised by N.
        # S = 0 naturally if no edges were opened (c could still be N, one per vertex,
        # but each component has size 1, so maximum/N -> 1/N, approaching 0 for large N).
        # The special case isempty(cnts) guards against c=0, which happens if N=0.
        cnts=zeros(Int,c); for v in 1:N; comp[v]>0&&(cnts[comp[v]]+=1); end
        isempty(cnts) ? 0.0 : maximum(cnts)/N
    end

    # ── Sweep over bond probability ───────────────────────────────────────
    # For each p in p_vals, run n_rep independent percolation trials and compute
    # two observables: S(p) (order parameter) and chi(p) (susceptibility).
    # Using multiple realisations at each p is necessary because percolation is
    # stochastic: a single trial has large variance near p_c, especially for small N.
    function _sweep(cx, p_vals, dim, n_rep)
        # S(p) = <s>: the sample mean of the giant-component fraction over n_rep seeds.
        # Seeds 1..n_rep are used so each realisation is statistically independent
        # (the same seeds are reused at every p, which is fine: the RNG is seeded
        # per call, so seed=1 at p=0.3 and seed=1 at p=0.7 are independent draws).
        S   = [mean(_perc_once(cx,p,dim,s) for s in 1:n_rep) for p in p_vals]

        # chi(p) = N * Var(S) / <S>: a fluctuation-based proxy for percolation susceptibility.
        # The standard textbook susceptibility is chi = sum_{s != giant} s^2 * n_s / N,
        # the second moment of non-giant cluster sizes within a single realisation.
        # We instead compute order-parameter fluctuations across n_rep realisations
        # (analogous to N*Var(m) in the Ising model, with an extra 1/<S> normalisation
        # to regularise the deep subcritical regime where <S> -> 1/N).
        # This proxy peaks sharply near p_c and correctly locates the critical point,
        # but its exponent and amplitude differ from the standard second-moment formula.
        chi = begin
            N=cx.N
            [begin
                # Re-run n_rep trials at this p to get the sample distribution of s.
                # Note: _perc_once is called a second time here (not reusing the S
                # samples above). This is intentional: S and chi need the same set
                # of realisations so they are statistically paired, and pulling them
                # in one pass keeps the code readable at a modest extra cost.
                ss=[_perc_once(cx,p,dim,s) for s in 1:n_rep]
                ms=mean(ss)
                # Compute <s^2> directly from the sample to avoid accumulating
                # a separate running sum. ms2 - ms^2 = Var(s) by definition.
                ms2=mean(s^2 for s in ss)
                # Regularise the denominator by 1e-12 to avoid 0/0 deep in the
                # subcritical phase (p << p_c), where <s> ~ 1/N -> 0 and
                # Var(s) is also near zero. The regularisation shifts chi to ~0
                # in that regime rather than producing NaN.
                # max(0,...) clamps away negative values that floating-point
                # rounding can introduce when ms2 - ms^2 < 0 by tiny amounts.
                max(0.0,N*(ms2-ms^2)/(ms+1e-12))
             end for p in p_vals]
        end
        S, chi
    end

    nothing
end

PlutoTemplate.explanation(
    """
    The simulation begins by generating an Erdos-Renyi random graph G(N, p_edge): each of the C(N,2) \
    possible vertex pairs is independently included as an edge with probability p_edge. From this \
    underlying graph, a clique complex is built by promoting every complete subgraph to a simplex: \
    every triangle in the graph becomes a 2-simplex, and every K_4 subgraph becomes a 3-simplex, so \
    no additional randomness is needed beyond the original edge coin flips. With the complex fixed, \
    a bond-percolation sweep runs over occupation probabilities p from 0 to 1 in steps of 0.05. \
    At each p, a chosen class of simplices (edges for k=1, triangles for k=2, tetrahedra for k=3) is \
